METHOD / PHASE 3: NGINX REVERSE PROXY CONFIGURATION
--------------------------------------------------------------------
1. Install Nginx:
   sudo apt update && sudo apt install -y nginx

2. Create Nginx Configuration for Domain:
   cat << 'EOF' > /etc/nginx/sites-available/sagarsandha.com.np
   server {
       listen 80;
       server_name sagarsandha.com.np www.sagarsandha.com.np;

       client_max_body_size 50M;

       location / {
           proxy_pass http://127.0.0.1:3001;
           proxy_http_version 1.1;
           proxy_set_header Upgrade $http_upgrade;
           proxy_set_header Connection 'upgrade';
           proxy_set_header Host $host;
           proxy_set_header X-Real-IP $remote_addr;
           proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
           proxy_set_header X-Forwarded-Proto $scheme;
           proxy_cache_bypass $http_upgrade;
       }
   }
   EOF

3. Enable Site & Test Nginx:
   sudo ln -s /etc/nginx/sites-available/sagarsandha.com.np /etc/nginx/sites-enabled/
   sudo rm /etc/nginx/sites-enabled/default 2>/dev/null || true
   sudo nginx -t
   sudo systemctl reload nginx
