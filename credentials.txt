4. HOW TO RESET / REPLACE EXISTING ADMIN CREDENTIALS (Run on Server)
--------------------------------------------------------------------
Run on VPS server terminal (replace NEW_EMAIL and NEW_PASSWORD):

docker exec -i mountain-multimedia-web node -e '
const { betterAuth } = require("better-auth");
const { drizzleAdapter } = require("better-auth/adapters/drizzle");
const Database = require("better-sqlite3");
const { drizzle } = require("drizzle-orm/better-sqlite3");

const sqlite = new Database("/app/data/app.db");
const db = drizzle(sqlite);

const auth = betterAuth({
  database: drizzleAdapter(db, { provider: "sqlite" }),
  emailAndPassword: { enabled: true },
  secret: process.env.BETTER_AUTH_SECRET || "mountain_multimedia_secret_key_2026"
});

async function main() {
  const newEmail = "NEW_EMAIL@domain.com";
  const newPassword = "NEW_SECURE_PASSWORD";

  // Clear previous credentials
  sqlite.exec("DELETE FROM users; DELETE FROM accounts; DELETE FROM sessions;");

  const res = await auth.api.signUpEmail({
    body: {
      email: newEmail,
      password: newPassword,
      name: "Admin User"
    }
  });

  console.log("✅ Admin credentials updated successfully!");
  console.log("Email:", newEmail);
}

main();
'
